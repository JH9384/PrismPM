  axioms: [propext],

        | List.nil => (record SessionRetentionReferenceSet {values := (match (prim equal value head : Bool)
          | Bool.false => (match (retentionBytesLess value head)
            | Bool.false => (ctor List.cons [Bytes] head (ctor List.cons [Bytes] value (ctor List.nil [Bytes])))

          | Bool.true => (ctor List.cons [Bytes] head (ctor List.nil [Bytes])))})
        | List.cons unusedHead unusedTail => (match (prim slice objects 0 (prim quotient (prim length objects : Nat) 2 0 : Nat) : (Option (List Bytes)))
          | Option.none => (record SessionRetentionReferenceSet {values := (ctor List.nil [Bytes])})
          | Option.some left => (match (prim slice objects (prim quotient (prim length objects : Nat) 2 0 : Nat) (prim subtract (prim length objects : Nat) (prim quotient (prim length objects : Nat) 2 0 : Nat) : Nat) : (Option (List Bytes)))

                | Bool.false => (prim append left (project (retentionInsertRefTree right value prior) values) : (List Bytes))

  name: retentionDifferenceAt,
  parameters: [objects : (List Bytes), retire : (List Bytes), fuel : Nat],
  result: (Option (List Bytes)),
  body: (match fuel
    | Nat.zero => (ctor Option.none [(List Bytes)])
    | Nat.succ remaining => (match objects
      | List.nil => (ctor Option.some [(List Bytes)] (ctor List.nil [Bytes]))
      | List.cons head tail => (match tail
        | List.nil => (ctor Option.some [(List Bytes)] (match (retentionSubset (ctor List.cons [Bytes] head (ctor List.nil [Bytes])) retire)
          | Bool.false => (ctor List.cons [Bytes] head (ctor List.nil [Bytes]))
          | Bool.true => (ctor List.nil [Bytes])))
        | List.cons unusedHead unusedTail => (match (prim slice objects 0 (prim quotient (prim length objects : Nat) 2 0 : Nat) : (Option (List Bytes)))
          | Option.none => (ctor Option.none [(List Bytes)])
          | Option.some left => (match (prim slice objects (prim quotient (prim length objects : Nat) 2 0 : Nat) (prim subtract (prim length objects : Nat) (prim quotient (prim length objects : Nat) 2 0 : Nat) : Nat) : (Option (List Bytes)))
            | Option.none => (ctor Option.none [(List Bytes)])
            | Option.some right => (match (retentionDifferenceAt left retire remaining)
              | Option.none => (ctor Option.none [(List Bytes)])
              | Option.some keptLeft => (match (retentionDifferenceAt right retire remaining)
                | Option.none => (ctor Option.none [(List Bytes)])
                | Option.some keptRight => (ctor Option.some [(List Bytes)] (prim append keptLeft keptRight : (List Bytes)))))))))),
  axioms: [propext],
  recursive_argument: fuel
}

definition{
  name: retentionFinalize,
  parameters: [
    snapshot : SessionRetentionSnapshot,
    objects : (List Bytes),
    roots : (List SessionRetentionRoot)
  ],
  result: (Result SessionRetentionSnapshot SessionRetentionError),
  body: (match ((prim length objects : Nat) <=? 4096)
    | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.ObjectLimit))
    | Bool.true => (match ((prim length roots : Nat) <=? 64)
      | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.RootLimit))
      | Bool.true => (match (retentionRootsValid roots objects (ctor Option.none [String]) 64)
        | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.ProtectedOrMissingObject))
        | Bool.true => (ctor Result.ok [SessionRetentionSnapshot, SessionRetentionError] (record SessionRetentionSnapshot {version := 1, revision := ((project snapshot revision) + 1), objects := objects, roots := roots}))))),
  axioms: [propext]

definition{
  name: commitSessionRetention,
  parameters: [
    snapshot : SessionRetentionSnapshot,
    replacement : (Option SessionRetentionReplacement),
    additions : (List Bytes),
    retire : (List Bytes)
  ],
  result: (Result SessionRetentionSnapshot SessionRetentionError),
  body: (match (sessionRetentionSnapshotValid snapshot)
    | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.BadSnapshot))
    | Bool.true => (match (retentionOptionalReplacementValid replacement)
      | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.BadReplacement))
      | Bool.true => (match (retentionRefsValid additions 16)
        | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.BadAdditions))
        | Bool.true => (match (retentionRefsValid retire 4096)
          | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.BadRetirement))
          | Bool.true => (match ((project snapshot revision) <? 4294967295)
            | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.RevisionExhausted))
            | Bool.true => (match (retentionExpectedHeadFits snapshot replacement)
              | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.ExpectedHeadMismatch))
              | Bool.true => (match (retentionSubset retire (project snapshot objects))
                | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.MissingRetirement))
                | Bool.true => (match (retentionDisjoint additions retire)
                  | Bool.false => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.AdditionRetirementOverlap))
                  | Bool.true => (match (retentionDifference (project (retentionUnion (project snapshot objects) additions) values) retire)
                    | Option.none => (ctor Result.error [SessionRetentionSnapshot, SessionRetentionError] (ctor SessionRetentionError.InternalBudget))
                    | Option.some objects => (retentionFinalize snapshot objects (project (retentionAfterRoots snapshot replacement) values))))))))))),
  axioms: [propext]
}
