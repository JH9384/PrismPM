definition{
  name: mailboxCheckExistingBinding,
  parameters: [state : MailboxAccountState, proof : AdmittedMailboxAssertion],
  result: (Result MailboxAdmissionAccepted MailboxAdmissionError),
  body: (match (project state mailboxBinding)
    | Option.none => (match (prim equal (mailboxOperationCode (project (project proof intent) operation)) (mailboxOperationCode (ctor MailboxOperation.Enroll)) : Bool)
      | Bool.false => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.UnboundAccount))
      | Bool.true => (mailboxAccept state proof))
    | Option.some binding => (match (project (project proof intent) operation)
      | MailboxOperation.Enroll => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.AlreadyBound))
      | MailboxOperation.Login => (match ((prim equal (project binding issuer) (project (project (project proof intent) authority) issuer) : Bool) && ((prim equal (project binding installationRef) (project (project (project proof intent) authority) installationRef) : Bool) && true))
        | Bool.false => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.WrongAuthority))
        | Bool.true => (match (prim equal (project binding mailbox) (project (project proof intent) mailbox) : Bool)
          | Bool.false => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.WrongMailbox))
          | Bool.true => (match (prim equal (project binding subject) (project proof subject) : Bool)
            | Bool.false => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.WrongSubject))
            | Bool.true => (mailboxAccept state proof))))
      | MailboxOperation.Recover => (match ((prim equal (project binding issuer) (project (project (project proof intent) authority) issuer) : Bool) && ((prim equal (project binding installationRef) (project (project (project proof intent) authority) installationRef) : Bool) && true))
        | Bool.false => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.WrongAuthority))
        | Bool.true => (match (prim equal (project binding mailbox) (project (project proof intent) mailbox) : Bool)
          | Bool.false => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.WrongMailbox))
          | Bool.true => (match (prim equal (project binding subject) (project proof subject) : Bool)
            | Bool.false => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.WrongSubject))
            | Bool.true => (match (not (prim equal (project state credentialRef) (project (project proof intent) candidateCredentialRef) : Bool))
              | Bool.false => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.NoChange))
              | Bool.true => (mailboxAccept state proof)))))
      | MailboxOperation.ReplaceMailbox => (match (mailboxBindingEqual binding (mailboxProofBinding proof))
        | Bool.false => (mailboxAccept state proof)
        | Bool.true => (ctor Result.error [MailboxAdmissionAccepted, MailboxAdmissionError] (ctor MailboxAdmissionError.NoChange))))),
  axioms: []
}
