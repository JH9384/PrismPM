    | Option.none => false
    | Option.some ySquared => (match (p256FieldMultiply x x)
      | Option.none => false
      | Option.some xSquared => (match (p256FieldMultiply xSquared x)
        | Option.none => false

            | Option.none => false

              | Option.none => false
              | Option.some minusThreeX => (match (p256FieldAdd minusThreeX (p256CurveB))
                | Option.none => false
                | Option.some rightSide => ((p256DigitsOrder ySquared rightSide) =? 0)))))))),
