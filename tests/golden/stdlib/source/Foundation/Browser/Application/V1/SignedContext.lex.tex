  name: signedContextSigningContext,

  result: String,

definition{
  name: signedContextKeyValid,
  parameters: [key : Bytes, purpose : Nat],
  result: Bool,
  body: ((Foundation.Browser.Application.V1.Effects::effectSigningGrantValid (record Foundation.Browser.Application.V1.Effects::EffectSigningGrant {publicKey := key, context := (signedContextSigningContext purpose)})) && (Foundation.Crypto.P256.Model::p256PublicKeyValid key)),
  axioms: [Quot.sound, propext]
}

definition{
  name: signedContextEnvelopeValid,
  parameters: [value : SignedContextEnvelope],
  result: Bool,
  body: ((signedContextValid (project value context)) && ((signedContextKeyValid (project value publicKey) (project (project value context) purpose)) && (((prim length (project value signature) : Nat) =? 64) && true))),
  axioms: [Quot.sound, propext]
}

definition{
  name: matchSignedContext,
  parameters: [expected : SignedContext, key : Bytes, envelope : SignedContextEnvelope],
  result: (Result Bool SignedContextError),
  body: (match ((signedContextValid expected) && ((signedContextValid (project envelope context)) && true))
    | Bool.false => (ctor Result.error [Bool, SignedContextError] (ctor SignedContextError.BadContext))
    | Bool.true => (match ((signedContextKeyValid key (project expected purpose)) && ((signedContextKeyValid (project envelope publicKey) (project (project envelope context) purpose)) && true))
      | Bool.false => (ctor Result.error [Bool, SignedContextError] (ctor SignedContextError.BadKey))
      | Bool.true => (match ((prim length (project envelope signature) : Nat) =? 64)
        | Bool.false => (ctor Result.error [Bool, SignedContextError] (ctor SignedContextError.BadSignature))
        | Bool.true => (match (signedContextEqual expected (project envelope context))
          | Bool.false => (ctor Result.error [Bool, SignedContextError] (ctor SignedContextError.ContextMismatch))
          | Bool.true => (match (prim equal key (project envelope publicKey) : Bool)
            | Bool.false => (ctor Result.error [Bool, SignedContextError] (ctor SignedContextError.KeyMismatch))
            | Bool.true => (ctor Result.ok [Bool, SignedContextError] true)))))),
  axioms: [Quot.sound, propext]
}
