definition{
  name: budgetAdapterFits,
  parameters: [adapter : Foundation.Browser.Application.V1.Effects::EffectAdapter, maximum : Nat],
  result: Bool,
  body: (match adapter
    | Foundation.Browser.Application.V1.Effects::EffectAdapter.Guest guest => (prim equal maximum (project guest inputMaximum) : Bool)
    | Foundation.Browser.Application.V1.Effects::EffectAdapter.Random => ((0 <? maximum) && ((maximum <=? 65536) && true))
    | Foundation.Browser.Application.V1.Effects::EffectAdapter.Digest => ((0 <? maximum) && ((maximum <=? 1048576) && true))
    | Foundation.Browser.Application.V1.Effects::EffectAdapter.Signer signing => ((0 <? maximum) && ((maximum <=? 1048576) && true))
    | Foundation.Browser.Application.V1.Effects::EffectAdapter.Verifier verification => ((0 <? maximum) && ((maximum <=? 1048576) && true))
    | Foundation.Browser.Application.V1.Effects::EffectAdapter.Store storage => (prim equal maximum (project storage maxObjectBytes) : Bool)),
  axioms: []
}

definition{
  name: budgetRowsFit,
  parameters: [
    items : (List ResourceBudget),
    grants : (List Foundation.Browser.Application.V1.Effects::EffectGrant),
    previous : String,
    fuel : Nat
  ],
  result: Bool,
  body: (match fuel
    | Nat.zero => (match items
      | List.nil => (match grants | List.nil => true | List.cons grant other => false)
      | List.cons head tail => false)
    | Nat.succ rest => (match items
      | List.nil => (match grants | List.nil => true | List.cons grant other => false)
      | List.cons head tail => (match grants
        | List.nil => false
        | List.cons grant other => ((prim equal (project head resource) (project grant resource) : Bool) && ((Foundation.Browser.Application.V1.Custody::custodyTextLess previous (project head resource)) && ((budgetAdapterFits (project grant adapter) (project head maximum)) && ((budgetRowsFit tail other (project head resource) rest) && true))))))),
  axioms: [],
  recursive_argument: fuel
}

  name: budgetOperationFits,
  parameters: [
    operation : Foundation.Browser.Application.V1.Effects::EffectOperation,
    maximum : Nat

  result: Bool,
  body: (match operation
    | Foundation.Browser.Application.V1.Effects::EffectOperation.InvokeGuest invocation => ((prim length (project invocation input) : Nat) <=? maximum)
    | Foundation.Browser.Application.V1.Effects::EffectOperation.RandomBytes amount => (amount <=? maximum)
    | Foundation.Browser.Application.V1.Effects::EffectOperation.DigestBytes payload => ((prim length payload : Nat) <=? maximum)
    | Foundation.Browser.Application.V1.Effects::EffectOperation.SignBytes payload => ((prim length payload : Nat) <=? maximum)
    | Foundation.Browser.Application.V1.Effects::EffectOperation.VerifyBytes verification => ((prim length (project verification value) : Nat) <=? maximum)
    | Foundation.Browser.Application.V1.Effects::EffectOperation.ReadObject digest => true
    | Foundation.Browser.Application.V1.Effects::EffectOperation.ReadHead name => true
    | Foundation.Browser.Application.V1.Effects::EffectOperation.Commit commit => (Foundation.Browser.Application.V1.Effects::effectObjectsValid (project commit objects) maximum)),
  axioms: []

definition{
  name: budgetRequestResourceFits,
  parameters: [
    items : (List ResourceBudget),
    request : Foundation.Browser.Application.V1.Effects::EffectRequest,
    fuel : Nat

  result: Bool,
  body: (match fuel
