  name: equalStrings,
  parameters: [left : (List String), right : (List String)],
  result: Bool,
  body: (match left
    | List.nil => (match right | List.nil => true | List.cons h t => false)

      | List.nil => false
      | List.cons other rest => ((prim equal head other : Bool) && ((equalStrings tail rest) && true)))),
  recursive_argument: left
}
