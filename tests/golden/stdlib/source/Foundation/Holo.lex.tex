definition{
  name: allConsecutive,
  parameters: [expected : Nat, values : (List Nat)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons value rest => ((expected =? value) && (allConsecutive (expected + 1) rest))),
  recursive_argument: values
}

definition{
  name: allConsecutiveProp,
  parameters: [expected : Nat, values : (List Nat)],
  result: Prop,
  body: (match values
    | List.nil => (expected = expected)
    | List.cons value rest => ((expected = value) /\ (allConsecutiveProp (expected + 1) rest))),
  recursive_argument: values
}

definition{
  name: allBelow,
  parameters: [bound : Nat, values : (List Nat)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons value rest => ((value <? bound) && (allBelow bound rest))),
  recursive_argument: values
}

definition{
  name: allBelowProp,
  parameters: [bound : Nat, values : (List Nat)],
  result: Prop,
  body: (match values
    | List.nil => (bound = bound)
    | List.cons value rest => ((value < bound) /\ (allBelowProp bound rest))),
  recursive_argument: values
}

theorem{
  name: allConsecutive_sound_complete,
  parameters: [expected : Nat, values : (List Nat)],
  statement: (((allConsecutive expected values) = true) <-> (allConsecutiveProp expected values)),
  proof: boolean_reflection{
    reflection: list{
      boolean_definition: allConsecutive,
      comparison: nat_beq,
      parameter: expected,
      proposition_definition: allConsecutiveProp,
      values: values
    }
  }
}
