definition{
  name: relationAllConsecutive,
  parameters: [expected : Nat, values : (List Nat)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons value rest => ((expected =? value) && (relationAllConsecutive (expected + 1) rest))),
  recursive_argument: values
}

definition{
  name: relationAllConsecutiveProp,
  parameters: [expected : Nat, values : (List Nat)],
  result: Prop,
  body: (match values
    | List.nil => (expected = expected)
    | List.cons value rest => ((expected = value) /\ (relationAllConsecutiveProp (expected + 1) rest))),
  recursive_argument: values
}

definition{
  name: relationAllBelow,
  parameters: [bound : Nat, values : (List Nat)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons value rest => ((value <? bound) && (relationAllBelow bound rest))),
  recursive_argument: values
}

definition{
  name: relationAllBelowProp,
  parameters: [bound : Nat, values : (List Nat)],
  result: Prop,
  body: (match values
    | List.nil => (bound = bound)
    | List.cons value rest => ((value < bound) /\ (relationAllBelowProp bound rest))),
  recursive_argument: values
}

definition{
  name: relationAllPositive,
  parameters: [values : (List Nat)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons value rest => ((0 <? value) && (relationAllPositive rest))),
  recursive_argument: values
}

theorem{
  name: relationAllConsecutive_sound_complete,
  parameters: [expected : Nat, values : (List Nat)],
  statement: (((relationAllConsecutive expected values) = true) <-> (relationAllConsecutiveProp expected values)),
  proof: boolean_reflection{
    reflection: list{
      boolean_definition: relationAllConsecutive,
      comparison: nat_beq,
      parameter: expected,
      proposition_definition: relationAllConsecutiveProp,
      values: values
    }
  }
}

theorem{
  name: relationAllBelow_sound_complete,
  parameters: [bound : Nat, values : (List Nat)],
  statement: (((relationAllBelow bound values) = true) <-> (relationAllBelowProp bound values)),
  proof: boolean_reflection{
    reflection: list{
      boolean_definition: relationAllBelow,
      comparison: nat_blt,
      parameter: bound,
      proposition_definition: relationAllBelowProp,
      values: values
    }
  }
}

definition{
  name: nonEmptyNatList,
  parameters: [values : (List Nat)],
  result: Bool,
  body: (match values | List.nil => false | List.cons value rest => true),
  recursive_argument: values
}
