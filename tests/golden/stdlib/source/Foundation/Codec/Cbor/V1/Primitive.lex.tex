  name: cborWritePrimitiveChecked,
  parameters: [value : CborValue, limits : CborLimits],
  result: (Result Bytes CborError),
  body: (match value
    | CborValue.Unsigned integer => (cborWriteHead integer 0 limits)
    | CborValue.ByteString payload => (cborWritePayload payload 2 (project limits maximumBytes) limits)
    | CborValue.TextString textValue => (match ((cborTextLength textValue) <=? (project limits maximumText))
      | Bool.false => (ctor Result.error [Bytes, CborError] (ctor CborError.ValueLimit))
      | Bool.true => (cborWritePayload (prim utf8_encode textValue : Bytes) 3 (project limits maximumText) limits))
    | CborValue.Boolean boolean => (match (1 <=? (project limits maximumOutput))
      | Bool.false => (ctor Result.error [Bytes, CborError] (ctor CborError.ValueLimit))
      | Bool.true => (ctor Result.ok [Bytes, CborError] (match boolean
        | Bool.false => 0xf4
        | Bool.true => 0xf5)))
    | CborValue.Null => (match (1 <=? (project limits maximumOutput))
      | Bool.false => (ctor Result.error [Bytes, CborError] (ctor CborError.ValueLimit))
      | Bool.true => (ctor Result.ok [Bytes, CborError] 0xf6))),
