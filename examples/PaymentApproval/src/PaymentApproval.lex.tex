definition{
  name: parseRole,
  parameters: [value : String],
  result: (Option Role),
  body: (if (prim equal value "clerk" : Bool) then (ctor Option.some [Role] (ctor Role.Clerk)) else (if (prim equal value "manager" : Bool) then (ctor Option.some [Role] (ctor Role.Manager)) else (ctor Option.none [Role])))
}

definition{
  name: parseAmount,
  parameters: [value : String],
  result: (Option Int64),
  body: (match (prim split_exact value "-" (2 : UInt32) : (Option (List String)))
    | Option.none => (ctor Option.none [Int64])
    | Option.some parts => (match parts
      | List.nil => (ctor Option.none [Int64])
      | List.cons digits rest => (match rest
        | List.cons afterSign more => (ctor Option.none [Int64])
        | List.nil => (match (prim parse_decimal digits : (Option Int64))
          | Option.none => (ctor Option.none [Int64])
          | Option.some number => (ctor Option.some [Int64] number))))),
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: decodeRequest,
  parameters: [roleText : String, amountText : String, limitText : String],
  result: (Option Request),
  body: (match (parseRole roleText)
    | Option.none => (ctor Option.none [Request])
    | Option.some role => (if (prim equal amountText "0" : Bool) then (ctor Option.none [Request]) else (match (parseAmount amountText)
      | Option.none => (ctor Option.none [Request])
      | Option.some amount => (match (parseAmount limitText)
        | Option.none => (ctor Option.none [Request])
        | Option.some limit => (ctor Option.some [Request] (record Request {role := role, amount := amount, limit := limit})))))),
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: decodeFields,
  parameters: [fields : (List String)],
  result: (Option Request),
  body: (match fields
    | List.nil => (ctor Option.none [Request])
    | List.cons roleText afterRole => (match afterRole
      | List.nil => (ctor Option.none [Request])
      | List.cons amountText afterAmount => (match afterAmount
        | List.nil => (ctor Option.none [Request])
        | List.cons limitText afterLimit => (match afterLimit
          | List.nil => (decodeRequest roleText amountText limitText)
          | List.cons extra rest => (ctor Option.none [Request]))))),
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: judge,
  parameters: [request : Request],
  result: Decision,
  body: (match (project request role)
    | Role.Manager => (ctor Decision.Approved)
    | Role.Clerk => (match (prim checked_subtract (project request limit) (project request amount) : (Option Int64))
      | Option.none => (ctor Decision.Refused)
      | Option.some headroom => (match (prim split_exact (prim format_decimal headroom : String) "-" (2 : UInt32) : (Option (List String)))
        | Option.none => (ctor Decision.Refused)
        | Option.some parts => (match parts
          | List.nil => (ctor Decision.Refused)
          | List.cons digits rest => (match rest
            | List.nil => (ctor Decision.Approved)
            | List.cons afterSign more => (ctor Decision.Refused)))))),
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: encodeDecision,
  parameters: [decision : Decision],
  result: String,
  body: (match decision
    | Decision.Approved => "approved"
    | Decision.Refused => "refused"
    | Decision.Invalid => "invalid")
}

definition{
  name: dispatchString,

      | Option.some request => (encodeDecision (judge request)))),
