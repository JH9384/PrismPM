  parameters: [entries : (List OrganizationEntry)],

definition{
  name: createOrganization,
  parameters: [state : OrganizationRegistry, account : String, request : OrganizationCreation],
  result: (Result OrganizationRegistry OrganizationLifecycleError),
  body: (match (lifecycleRegistryValid state)
    | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.BadRegistry))
    | Bool.true => (match ((Foundation.Organization.V1.Administration::administrationBoundedText account 128) && ((Foundation.Organization.V1.Administration::administrationBoundedText (project request registry) 128) && ((Foundation.Organization.V1.Administration::administrationBoundedText (project request actor) 128) && (((project request revision) <=? 4294967295) && true))))
      | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.BadRequest))
      | Bool.true => (match (prim equal (project state registry) (project request registry) : Bool)
        | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.WrongRegistry))
        | Bool.true => (match ((project state revision) =? (project request revision))
          | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.StaleRevision))
          | Bool.true => (match ((project state revision) <? 4294967295)
            | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.RevisionExhausted))
            | Bool.true => (match (prim equal account (project request actor) : Bool)
              | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.ActorMismatch))
              | Bool.true => (match ((Foundation.Organization.V1.Administration::administrationBoundedText (project request request) 128) && ((Foundation.Organization.V1.Administration::administrationBoundedText (project request organization) 128) && ((Foundation.Organization.V1.Administration::administrationBoundedText (project request root) 128) && ((Foundation.Organization.V1.Administration::administrationBoundedText (project request label) 256) && true))))
                | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.BadRequest))
                | Bool.true => (match (not (lifecycleHasOrganization (project state entries) (project request organization)))
                  | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.DuplicateOrganization))
                  | Bool.true => (match (not (lifecycleHasCreationRequest (project state entries) (project request request)))
                    | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.DuplicateCreationRequest))
                    | Bool.true => (match ((lifecycleEntryCount (project state entries)) <? (project state capacity))
                      | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.RegistryFull))
                      | Bool.true => (match (Foundation.Organization.V1.Administration::createAdministration (project request organization) (project request label) account (project request root))
                        | Result.error error => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.AdministrationRejected error))
                        | Result.ok administration => (ctor Result.ok [OrganizationRegistry, OrganizationLifecycleError] (record OrganizationRegistry {registry := (project state registry), revision := ((project state revision) + 1), capacity := (project state capacity), entries := (project (lifecycleAppend (project state entries) (record OrganizationEntry {creator := account, creationRequest := (project request request), administration := administration})) entries)}))))))))))))),
  axioms: []
}

definition{
  name: administerOrganization,
  parameters: [
    state : OrganizationRegistry,
    account : String,
    request : OrganizationAdministration
  ],
  result: (Result OrganizationRegistry OrganizationLifecycleError),
  body: (match (lifecycleRegistryValid state)
    | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.BadRegistry))
    | Bool.true => (match ((Foundation.Organization.V1.Administration::administrationBoundedText account 128) && ((Foundation.Organization.V1.Administration::administrationBoundedText (project request registry) 128) && ((Foundation.Organization.V1.Administration::administrationBoundedText (project request actor) 128) && (((project request revision) <=? 4294967295) && true))))
      | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.BadRequest))
      | Bool.true => (match (prim equal (project state registry) (project request registry) : Bool)
        | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.WrongRegistry))
        | Bool.true => (match ((project state revision) =? (project request revision))
          | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.StaleRevision))
          | Bool.true => (match ((project state revision) <? 4294967295)
            | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.RevisionExhausted))
            | Bool.true => (match (prim equal account (project request actor) : Bool)
              | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.ActorMismatch))
              | Bool.true => (match (Foundation.Organization.V1.Administration::administrationBoundedText (project request organization) 128)
                | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.BadRequest))
                | Bool.true => (match (prim equal (project request organization) (project (project request administration) organization) : Bool)
                  | Bool.false => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.WrongOrganization))
                  | Bool.true => (match (lifecycleFind (project state entries) (project request organization))
                    | Option.none => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.UnknownOrganization))
                    | Option.some entry => (match (lifecycleApplyAdministration (project entry administration) (project request administration))
                      | Result.error error => (ctor Result.error [OrganizationRegistry, OrganizationLifecycleError] (ctor OrganizationLifecycleError.AdministrationRejected error))
                      | Result.ok next => (ctor Result.ok [OrganizationRegistry, OrganizationLifecycleError] (record OrganizationRegistry {registry := (project state registry), revision := ((project state revision) + 1), capacity := (project state capacity), entries := (project (lifecycleReplace (project state entries) (project request organization) next) entries)})))))))))))),
  axioms: []
}
